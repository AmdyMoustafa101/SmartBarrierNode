#include <Servo.h>

// Définir les broches
const int trigPin = 2;     // Broche TRIG du capteur ultrason
const int echoPin = 3;     // Broche ECHO du capteur ultrason
const int servo1Pin = 9;   // Broche du servomoteur 1
const int servo2Pin = 10;  // Broche du servomoteur 2
const int buzzerPin = 7;   // Broche du buzzer
const int buttonPin = 4;   // Broche du bouton

// Configuration communication
const int BUFFER_SIZE = 32;
char serialBuffer[BUFFER_SIZE];
int bufferIndex = 0;

// Seuil de détection (en cm)
const int distanceThreshold = 30;

// Créer les objets Servo
Servo servo1;
Servo servo2;

// Variables d'état
bool isWaitingForCar = false;
int receivedCommand = -1;
bool lastButtonState = LOW;

void setup() {
  Serial.begin(9600);
  
  // Configuration des broches
  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);
  pinMode(buzzerPin, OUTPUT);
  pinMode(buttonPin, INPUT_PULLUP);

  // Attacher les servomoteurs
  servo1.attach(servo1Pin);
  servo2.attach(servo2Pin);

  // Position initiale
  servo1.write(0);
  servo2.write(0);
}

void loop() {
  // Lecture du bouton
  bool currentButtonState = digitalRead(buttonPin);
  if (currentButtonState == HIGH && lastButtonState == LOW) {
    servo2.write(0);
    beep(600, 200);
    Serial.println("Bouton pressé - Servo 2 OUVERT");
    delay(50);
  }
  lastButtonState = currentButtonState;

  // Lecture série
  while (Serial.available() > 0) {
    char c = Serial.read();
    
    // Gestion du buffer circulaire
    if (bufferIndex < BUFFER_SIZE - 1) {
      serialBuffer[bufferIndex++] = c;
    }
    
    // Détection de fin de commande
    if (c == '\n') {
      serialBuffer[bufferIndex] = '\0'; // Terminaison de chaîne
      processCommand(serialBuffer);
      bufferIndex = 0; // Reset buffer
    }
  }

  // Gestion de la fermeture automatique du servo1
  if (isWaitingForCar) {
    float distance = getDistance();
   
    if (distance > distanceThreshold) {
      servo1.write(0);
      beep(500, 300);
      Serial.println("Voiture partie - Servo 1 fermé");
      isWaitingForCar = false;
    }
    delay(100);
  }
}

void processCommand(char* buffer) {
  // Vérification du format "CMDx"
  if (strncmp(buffer, "CMD", 3) == 0) {
    int command = atoi(buffer + 3); // Convertit le nombre après CMD
    
    if (command == 1) {
      servo1.write(90);
      beep(1000, 200);
      Serial.println("Servo 1 ouvert");
      delay(700);
      isWaitingForCar = true;
    }
    else if (command == 2) {
      servo2.write(90);
      beep(800, 100);
      delay(100);
      beep(800, 100);
      Serial.println("Servo 2 fermé");
    }
    else if (command == 0) {
      // Commande spéciale pour reset
      servo1.write(0);
      servo2.write(0);
      beep(900, 500);
      delay(500);
      beep(900, 500);
      Serial.println("Reset tous servos");
    }
  }
}

float getDistance() {
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);

  long duration = pulseIn(echoPin, HIGH);
  return duration * 0.034 / 2;
}

void beep(int frequency, int duration) {
  tone(buzzerPin, frequency);
  delay(duration);
  noTone(buzzerPin);
}
